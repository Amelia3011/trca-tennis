    <script>
        const PRICING = {
            'Jalan Kuching, KL': {
                'Court Kickstart': 45,
                'Core Court Membership': 140
            },
            'The Hood, KL': {
                'Court Kickstart': 50,
                'Core Court Membership': 160
            }
        };

        window.onload = function() {
            lucide.createIcons();
            updateBookingData();
        };

        function selectLocationAndScroll(locName) {
            const locRadios = document.getElementsByName('location');
            for (let r of locRadios) {
                if (r.value === locName) {
                    r.checked = true;
                }
            }
            updateBookingData();
            document.getElementById('booking-section').scrollIntoView({ behavior: 'smooth' });
        }

        function goToStep(stepNumber) {
            document.getElementById('step-1').classList.add('hidden');
            document.getElementById('step-2').classList.add('hidden');
            document.getElementById('step-3').classList.add('hidden');

            document.getElementById(`step-${stepNumber}`).classList.remove('hidden');

            const progressBar = document.getElementById('progress-bar');
            const stepLabel = document.getElementById('step-label');
            const stepPercent = document.getElementById('step-percent');

            if (stepNumber === 1) {
                progressBar.style.width = '33%';
                stepLabel.innerText = 'Step 1 of 3: Location & Package';
                stepPercent.innerText = '33% Completed';
            } else if (stepNumber === 2) {
                progressBar.style.width = '66%';
                stepLabel.innerText = 'Step 2 of 3: Age Group & Schedule';
                stepPercent.innerText = '66% Completed';
            } else if (stepNumber === 3) {
                progressBar.style.width = '100%';
                stepLabel.innerText = 'Step 3 of 3: Child & Parent Details';
                stepPercent.innerText = '100% Completed';
            }
        }

        function updateBookingData() {
            const selectedLocation = document.querySelector('input[name="location"]:checked')?.value || 'Jalan Kuching, KL';
            const selectedPlan = document.querySelector('input[name="plan"]:checked')?.value || 'Court Kickstart';
            const selectedSlot = document.querySelector('input[name="timeslot"]:checked')?.value || 'Saturday 9:00 AM';

            const kickstartPrice = PRICING[selectedLocation]['Court Kickstart'];
            const membershipPrice = PRICING[selectedLocation]['Core Court Membership'];

            document.getElementById('kickstart-price').innerText = `RM ${kickstartPrice}`;
            document.getElementById('membership-price').innerText = `RM ${membershipPrice}`;

            const activePrice = PRICING[selectedLocation][selectedPlan];

            document.getElementById('summary-location').innerText = selectedLocation;
            document.getElementById('summary-plan').innerText = selectedPlan;
            document.getElementById('summary-slot').innerText = selectedSlot;
            document.getElementById('summary-total').innerText = `RM ${activePrice}`;
        }

        function handleFormSubmit(event) {
            event.preventDefault();

            const childName = document.getElementById('childName').value;
            const selectedLocation = document.querySelector('input[name="location"]:checked')?.value;
            const selectedPlan = document.querySelector('input[name="plan"]:checked')?.value;
            const selectedSlot = document.querySelector('input[name="timeslot"]:checked')?.value;
            const activePrice = PRICING[selectedLocation][selectedPlan];

            const refNo = 'RCA-' + Math.floor(10000 + Math.random() * 90000);

            document.getElementById('res-ref').innerText = refNo;
            document.getElementById('res-child').innerText = childName;
            document.getElementById('res-loc').innerText = selectedLocation;
            document.getElementById('res-plan').innerText = selectedPlan;
            document.getElementById('res-slot').innerText = selectedSlot;
            document.getElementById('res-price').innerText = `RM ${activePrice}`;

            document.getElementById('confirmationModal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('confirmationModal').classList.add('hidden');
            goToStep(1);
            document.getElementById('bookingForm').reset();
            updateBookingData();
        }

        function toggleFaq(faqId) {
            const content = document.getElementById(faqId);
            const icon = document.getElementById(`icon-${faqId}`);
            
            if (content.classList.contains('hidden')) {
                content.classList.remove('hidden');
                icon.style.transform = 'rotate(180deg)';
            } else {
                content.classList.add('hidden');
                icon.style.transform = 'rotate(0deg)';
            }
        }
